.ORIG x2000

; ----------------------------------------
; Save registers used by ISR
; PC and PSR are automatically saved by hardware
; ----------------------------------------
        ADD R6, R6, #-1
        STR R0, R6, #0

        ADD R6, R6, #-1
        STR R1, R6, #0

        ADD R6, R6, #-1
        STR R2, R6, #0

        ADD R6, R6, #-1
        STR R3, R6, #0

        ADD R6, R6, #-1
        STR R4, R6, #0


; ----------------------------------------
; R4 points to input buffer x4000
; Read characters until Enter (x000A)
; ----------------------------------------
        LD  R4, STRING

READ_LOOP
        LDI R0, KBSR        ; read keyboard status register
        BRzp READ_LOOP      ; wait while keyboard is not ready

        LDI R0, KBDR        ; read one keyboard character

        LD  R2, NEG_ENTER
        ADD R2, R0, R2
        BRz INPUT_END       ; Enter pressed: stop receiving input

        STR R0, R4, #0      ; store ordinary character
        ADD R4, R4, #1
        BRnzp READ_LOOP


; ----------------------------------------
; Add x0000 as string terminator
; ----------------------------------------
INPUT_END
        AND R0, R0, #0
        STR R0, R4, #0


; ----------------------------------------
; Print the entire input string 10 times
; ----------------------------------------
        AND R3, R3, #0
        ADD R3, R3, #10

PRINT_10
        LD  R4, STRING      ; restart from beginning of string

PRINT_CHAR
        LDR R0, R4, #0
        BRz ONE_TIME_DONE   ; reach x0000: one copy finished

WAIT_DSR
        LDI R1, DSR         ; check display status register
        BRzp WAIT_DSR       ; wait until display is ready

        STI R0, DDR         ; output one character
        ADD R4, R4, #1
        BRnzp PRINT_CHAR


ONE_TIME_DONE
        ADD R3, R3, #-1
        BRp PRINT_10


; ----------------------------------------
; Restore registers
; ----------------------------------------
        LDR R4, R6, #0
        ADD R6, R6, #1

        LDR R3, R6, #0
        ADD R6, R6, #1

        LDR R2, R6, #0
        ADD R6, R6, #1

        LDR R1, R6, #0
        ADD R6, R6, #1

        LDR R0, R6, #0
        ADD R6, R6, #1

        RTI


; ----------------------------------------
; constants
; ----------------------------------------
KBSR        .FILL xFE00
KBDR        .FILL xFE02
DSR         .FILL xFE04
DDR         .FILL xFE06

NEG_ENTER   .FILL xFFF6     ; -x000A
STRING      .FILL x4000     ; input buffer begins at x4000

.END
